**Step 1: Create `prism-engine/src/routes/reports.ts`**:

```typescript
import { Hono } from 'hono';
import type { Env, ReportStatus } from '../lib/types';
import { withUser } from '../middleware/auth';
import { createReport, getWhitelistedSourceByUserId } from '../lib/queries';

export const reportRoutes = new Hono<{ Bindings: Env; Variables: import('../middleware/auth').AuthVariables }>();

const ALLOWED_MIME_TYPES = ['image/jpeg', 'image/png', 'image/webp'] as const;
const MAX_FILE_SIZE = 10 * 1024 * 1024; // 10MB per D-03

reportRoutes.post('/harvest', withUser(), async (c) => {
  const user = c.get('user');

  // D-02: Whitelisted source verification
  const source = await getWhitelistedSourceByUserId(c.env.DB, user.id);
  if (!source) {
    return c.json({ error: 'User is not a whitelisted source' }, 403);
  }

  // Parse multipart body
  const formData = await c.req.parseBody();

  // RPT-05: Validate media (D-03)
  const media = formData.get('media');
  if (!media || !(media instanceof File)) {
    return c.json({ error: 'Missing media file' }, 400);
  }

  if (!ALLOWED_MIME_TYPES.includes(media.type as any)) {
    return c.json({
      error: `Invalid media type: ${media.type}. Allowed: ${ALLOWED_MIME_TYPES.join(', ')}`,
    }, 400);
  }

  if (media.size > MAX_FILE_SIZE) {
    return c.json({
      error: `File too large: ${media.size} bytes. Max: ${MAX_FILE_SIZE} bytes`,
    }, 400);
  }

  // RPT-05: Validate coordinates
  const latStr = formData.get('latitude');
  const lonStr = formData.get('longitude');

  if (!latStr) {
    return c.json({ error: 'Missing latitude' }, 400);
  }
  if (!lonStr) {
    return c.json({ error: 'Missing longitude' }, 400);
  }

  const latitude = parseFloat(latStr as string);
  const longitude = parseFloat(lonStr as string);

  if (isNaN(latitude) || latitude < -90 || latitude > 90) {
    return c.json({ error: 'Invalid latitude: must be number between -90 and 90' }, 400);
  }
  if (isNaN(longitude) || longitude < -180 || longitude > 180) {
    return c.json({ error: 'Invalid longitude: must be number between -180 and 180' }, 400);
  }

  // RPT-03: Upload to R2 with UUID key
  const objectKey = `harvest/${crypto.randomUUID()}-${media.name}`;
  await c.env.VAULT.put(objectKey, await media.arrayBuffer());

  // RPT-02 + RPT-04: Create report (DIGIPIN auto-generated by createReport, status defaults to 'pending')
  const report = await createReport(c.env.DB, {
    reporterId: user.id,
    latitude,
    longitude,
    r2ImageUrl: `r2://${objectKey}`,
    status: 'pending',
  });

  return c.json({ report }, 201);
});
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client -> API | Untrusted input: Bearer token, multipart form data (media, coordinates) |
| Route -> R2 | R2 put with UUID key (no path traversal risk) |
| Route -> D1 | Parameterized queries via queries.ts |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-05 | Spoofing | POST /harvest auth | mitigate | withUser() middleware validates Bearer JWT via SuperTokens. Per-request D1 lookup. |
| T-03-06 | Tampering | Multipart media upload | mitigate | MIME type validation (image/jpeg, image/png, image/webp only per D-03). File size check (max 10MB per D-03). Reject invalid with 400. |
| T-03-07 | Tampering | Coordinate inputs | mitigate | Parse + range validation: latitude [-90,90], longitude [-180,180]. Reject invalid with 400. |
| T-03-08 | Elevation of privilege | Non-whitelisted user submitting | mitigate | D-02 check: query Whitelisted_Sources for approved record linked to user. Return 403 if missing. |
| T-03-09 | Denial of Service | Large file upload | mitigate | MAX_FILE_SIZE = 10MB. Check media.size after parseBody. Workers platform has 100MB body limit. |
| T-03-10 | Information disclosure | R2 object keys | accept | UUID-based keys (crypto.randomUUID()). Not guessable. R2 bucket not publicly accessible. |
</threat_model>
