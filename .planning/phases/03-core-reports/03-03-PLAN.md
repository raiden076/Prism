    /**
     * Get reports with RBAC filter + optional status filter + pagination.
     * whereClause/params come from getReportsFilter() in rbac middleware.
     */
    export async function getReportsWithFilter(
      db: D1Database,
      options: {
        whereClause: string;
        params: string[];
        status?: ReportStatus;
        limit?: number;
        offset?: number;
      }
    ): Promise<Report[]> {
      const { whereClause, params, status, limit = 100, offset = 0 } = options;

      let sql = `SELECT * FROM Reports WHERE ${whereClause}`;
      const allParams: (string | number)[] = [...params];

      if (status) {
        sql += ' AND status = ?';
        allParams.push(status);
      }

      sql += ' ORDER BY created_at DESC LIMIT ? OFFSET ?';
      allParams.push(limit, offset);

      const { results } = await db
        .prepare(sql)
        .bind(...allParams)
        .all<ReportRow>();
      return results.map(rowToReport);
    }
    ```

    /**
     * PRISM - Board Routes (Query + Status Transition)
     *
     * GET  /              - Board query: paginated reports with RBAC scope + optional status filter
     * GET  /nearby        - Nearby reports with Haversine distance
     * PATCH /:id/status   - Status transition with state machine enforcement
     */
    import { Hono } from 'hono';
    import type { Context } from 'hono';
    import type { Env } from '../lib/types';
    import type { AuthVariables } from '../middleware/auth';
    import { getReportsWithFilter, getNearbyReports, getReportById, updateReportStatus } from '../lib/queries';
    import { getReportsFilter } from '../middleware/rbac';
    import { REPORT_STATUSES, isValidTransition, STATUS_TRANSITIONS } from '../lib/types';
    import type { ReportStatus } from '../lib/types';

    export const boardRoutes = new Hono<{ Bindings: Env; Variables: AuthVariables }>();

    // GET / - Board query with RBAC + optional status filter + pagination
    boardRoutes.get('/', async (c) => {
      const user = c.get('user');

      const status = c.req.query('status') as ReportStatus | undefined;

      // Validate status param if provided
      if (status && !REPORT_STATUSES.includes(status)) {
        return c.json({
          error: `Invalid status filter. Valid values: ${REPORT_STATUSES.join(', ')}`
        }, 400);
      }

      const { whereClause, params } = await getReportsFilter(user.role, user.id, c.env.DB);

      const reports = await getReportsWithFilter(c.env.DB, {
        whereClause,
        params,
        status,
        limit: 100,
      });

      return c.json({ data: reports }, 200);
    });

    // GET /nearby - Nearby reports by lat/lon/radius with RBAC
    boardRoutes.get('/nearby', async (c) => {
      const user = c.get('user');
      const latStr = c.req.query('lat');
      const lonStr = c.req.query('lon');
      const radiusStr = c.req.query('radius') ?? '200';

      if (!latStr || !lonStr) {
        return c.json({ error: 'Missing lat/lon parameters' }, 400);
      }

      const latitude = parseFloat(latStr);
      const longitude = parseFloat(lonStr);
      const radius = parseFloat(radiusStr);

      if (isNaN(latitude) || isNaN(longitude) || isNaN(radius)) {
        return c.json({ error: 'Invalid lat/lon/radius values' }, 400);
      }

      const nearbyReports = await getNearbyReports(c.env.DB, latitude, longitude, radius);

      // Post-filter by RBAC: get allowed reporter IDs from filter
      const { whereClause, params } = await getReportsFilter(user.role, user.id, c.env.DB);

      // For admin (whereClause='1=1'), skip post-filter
      if (whereClause === '1=1') {
        return c.json({ data: nearbyReports }, 200);
      }

      // For non-admin: fetch all report IDs visible to user, then intersect with nearby
      const visibleReports = await getReportsWithFilter(c.env.DB, {
        whereClause,
        params,
        limit: 1000,
      });
      const visibleIds = new Set(visibleReports.map(r => r.id));
      const filtered = nearbyReports.filter(r => visibleIds.has(r.id));

      return c.json({ data: filtered }, 200);
    });

    // PATCH /:id/status - Status transition with state machine enforcement
    boardRoutes.patch('/:id/status', async (c) => {
      const user = c.get('user');

      const reportId = c.req.param('id');
      const body = await c.req.json();

      if (!body.status) {
        return c.json({ error: 'Missing status field in request body' }, 400);
      }

      const newStatus = body.status as ReportStatus;

      if (!REPORT_STATUSES.includes(newStatus)) {
        return c.json({
          error: `Invalid status. Valid values: ${REPORT_STATUSES.join(', ')}`
        }, 400);
      }

      const report = await getReportById(c.env.DB, reportId);
      if (!report) {
        return c.json({ error: 'Report not found' }, 404);
      }

      if (!isValidTransition(report.status, newStatus)) {
        return c.json({
          error: `Invalid transition: ${report.status} -> ${newStatus}`,
          validTransitions: STATUS_TRANSITIONS[report.status],
        }, 400);
      }

      const updated = await updateReportStatus(c.env.DB, reportId, newStatus);
      if (!updated) {
        return c.json({ error: 'Failed to update report status' }, 500);
      }

      return c.json({ data: updated }, 200);
    });
    ```
