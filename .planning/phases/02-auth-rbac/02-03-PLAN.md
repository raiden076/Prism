**Create implementation** `prism-engine/src/middleware/auth.ts`:

```typescript
import type { Context, Next } from 'hono';
import type { Env, User } from '../lib/types';
import { verifyAccessToken } from '../lib/supertokens-adapter';
import { getUserBySuperTokensId } from '../lib/queries';

export type AuthVariables = {
  user: User;
  supertokensUserId: string;
};

export function withUser() {
  return async (
    c: Context<{ Bindings: Env; Variables: AuthVariables }>,
    next: Next
  ) => {
    const authHeader = c.req.header('Authorization');
    if (!authHeader?.startsWith('Bearer ')) {
      return c.json({ error: 'Unauthorized -- missing or malformed Authorization header' }, 401);
    }

    const accessToken = authHeader.slice(7);
    const payload = await verifyAccessToken(accessToken, c.env.SUPERTOKENS_CORE_URL, c.env.SUPERTOKENS_API_KEY);
    if (!payload) {
      return c.json({ error: 'Unauthorized -- invalid or expired token' }, 401);
    }

    // Per-request D1 lookup (D-08) -- always fresh role data
    const user = await getUserBySuperTokensId(c.env.DB, payload.userId);
    if (!user) {
      return c.json({ error: 'User not found -- SuperTokens user has no D1 record' }, 404);
    }

    c.set('user', user);
    c.set('supertokensUserId', payload.userId);
    await next();
  };
}
```

**Create implementation** `prism-engine/src/middleware/rbac.ts`:

```typescript
import type { Context, Next } from 'hono';
import type { User, UserRole } from '../lib/types';
import { getUserDescendants } from '../lib/queries';

export function requireRole(...roles: UserRole[]) {
  return async (c: Context<{ Variables: { user: User } }>, next: Next) => {
    const user = c.get('user');
    if (!user) {
      return c.json({ error: 'Forbidden -- no user context. Ensure withUser() runs first.' }, 403);
    }
    if (!roles.includes(user.role)) {
      return c.json({ error: `Forbidden -- requires one of: ${roles.join(', ')}` }, 403);
    }
    await next();
  };
}

export async function getReportsFilter(
  role: UserRole,
  userId: string,
  db: D1Database
): Promise<{ whereClause: string; params: string[] }> {
  switch (role) {
    case 'admin':
      return { whereClause: '1=1', params: [] };

    case 'contractor':
      return {
        whereClause: 'id IN (SELECT report_id FROM Interventions WHERE contractor_id = ?)',
        params: [userId],
      };

    case 'crony':
      return {
        whereClause: '(reporter_id = ? OR id IN (SELECT report_id FROM Verifications WHERE verifier_id = ?))',
        params: [userId, userId],
      };

    default:
      // Hierarchy-scoped access (RBAC-05): users with non-standard roles
      // or hierarchy heads see reports from their subtree
      const descendants = await getUserDescendants(db, userId);
      const ids = descendants.map(d => d.id);
      if (ids.length === 0) {
        return { whereClause: '1=0', params: [] };
      }
      const placeholders = ids.map(() => '?').join(',');
      return {
        whereClause: `reporter_id IN (${placeholders})`,
        params: ids,
      };
  }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client -> Protected routes | Bearer token verified by withUser(). Role checked by requireRole(). No client-sent role trusted. |
| Middleware -> D1 | User role fetched from DB on every request (D-08). Never cached beyond single request. |
| RBAC filter -> SQL | Parameterized queries only. WHERE clauses use ? placeholders with .bind(). |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-11 | Elevation of Privilege | withUser() | mitigate | Role fetched from D1 on every request (D-08). Never trusts client-sent role. JWT only provides userId, role comes from DB. |
| T-02-12 | Elevation of Privilege | requireRole() | mitigate | Explicit role whitelist in middleware. Route must declare allowed roles. No wildcard access. |
| T-02-13 | Information Disclosure | getReportsFilter() | mitigate | Contractor filter uses subquery, never exposes other users' report IDs. Hierarchy filter uses server-side CTE, never trusts client hierarchy params. |
| T-02-14 | Spoofing | withUser() | mitigate | Bearer token format enforced (starts with "Bearer "). JWT verified via jose with JWKS key rotation. |
| T-02-15 | Tampering | getReportsFilter hierarchy | mitigate | Subtree IDs computed server-side via recursive CTE on supervisor_id. Client cannot inject or modify hierarchy traversal. |
</threat_model>
