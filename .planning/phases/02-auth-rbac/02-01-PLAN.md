```typescript
export async function getUserIdFromSession(
  request: Request,
  coreUrl: string,
  apiKey: string
): Promise<string | null> {
  const authHeader = request.headers.get('Authorization');
  if (!authHeader?.startsWith('Bearer ')) return null;
  const accessToken = authHeader.slice(7);
  const payload = await verifyAccessToken(accessToken, coreUrl, apiKey);
  return payload?.userId ?? null;
}
```

2. Fix `revokeSession` similarly:
```typescript
export async function revokeSession(
  request: Request,
  coreUrl: string,
  apiKey: string
): Promise<boolean> {
  const authHeader = request.headers.get('Authorization');
  if (!authHeader?.startsWith('Bearer ')) return false;
  const accessToken = authHeader.slice(7);
  const payload = await verifyAccessToken(accessToken, coreUrl, apiKey);
  if (!payload) return false;
  return revokeSession(payload.sessionHandle, coreUrl, apiKey);
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client -> Workers API | Untrusted bearer token in Authorization header. Must verify JWT signature + expiry before trusting claims. |
| Workers -> SuperTokens Core | API calls to Core carry api-key in header. Must use HTTPS, never log api-key. |
| Workers -> D1 | Parameterized queries only. No SQL interpolation from token claims. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Spoofing | verifyAccessToken | mitigate | JWT signature verification via jose with JWKS key rotation. Never trust raw claims without signature check. |
| T-02-02 | Tampering | Authorization header | mitigate | Bearer token format enforcement (starts with "Bearer "), reject malformed headers with 401 |
| T-02-03 | Information Disclosure | Core API calls | mitigate | Never log SUPERTOKENS_API_KEY. Error responses to client contain generic messages ("Unauthorized"), not internal details |
| T-02-04 | Tampering | OTP verification | mitigate | OTP codes verified server-side via Core API, never trusted from client claims |
| T-02-05 | Elevation of Privilege | upsertUserBySuperTokens | mitigate | New users always created with 'crony' role (D-09). Role escalation requires admin action, never auto-assigned |
</threat_model>
